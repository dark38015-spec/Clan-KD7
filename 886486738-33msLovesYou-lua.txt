    if newHum then hum = newHum end
end

            dragging=true; dragStart=inp.Position; startPos=MF.Position
        end
    end)
    UIS.InputChanged:Connect(function(inp)
        if dragging and inp.UserInputType==Enum.UserInputType.MouseMovement then
            local d = inp.Position - dragStart
            MF.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset+d.X,
                                    startPos.Y.Scale, startPos.Y.Offset+d.Y)
        end
    end)
    UIS.InputEnded:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseButton1 then dragging=false end
    end)
end

-- سحب دائرة التصغير
do
    local dragging, dragStart, startPos, moved = false, nil, nil, false
    MiniCircle.InputBegan:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseButton1 then
            dragging=true; moved=false; dragStart=inp.Position; startPos=MiniCircle.Position
        end
    end)
    UIS.InputChanged:Connect(function(inp)
        if dragging and inp.UserInputType==Enum.UserInputType.MouseMovement then
            local d = inp.Position - dragStart
            if d.Magnitude > 4 then moved=true end
            if moved then
                MiniCircle.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset+d.X,
                                                startPos.Y.Scale, startPos.Y.Offset+d.Y)
            end
        end
    end)
    UIS.InputEnded:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseButton1 and dragging then
            dragging=false
            if not moved then MiniCircle.Visible=false; MF.Visible=true end
        end
    end)
end

local tabs, pages = {}, {}
local function makeTab(name)
    local btn = Instance.new("TextButton", TabBar)
    btn.Size            = UDim2.new(0,80,0,20)
    btn.BackgroundColor3= Color3.fromRGB(38,38,38)
    btn.Text            = name
    btn.TextColor3      = Color3.fromRGB(190,190,190)
    btn.Font            = Enum.Font.Gotham
    btn.TextSize        = 11
    btn.BorderSizePixel = 0
    btn.ZIndex          = 4
    addCorner(btn, 5)
    return btn
end
local function makePage()
    local pg = Instance.new("Frame", CF)
    pg.Size                   = UDim2.new(1,0,1,0)
    pg.BackgroundTransparency = 1
    pg.BorderSizePixel        = 0
    pg.Visible                = false
    pg.ZIndex                 = 4
    return pg
end
local function showPage(name)
    for k,pg in pairs(pages) do pg.Visible=(k==name) end
    for k,tb in pairs(tabs) do
        tb.BackgroundColor3 = (k==name) and Color3.fromRGB(65,65,65) or Color3.fromRGB(38,38,38)
        tb.TextColor3       = (k==name) and Color3.fromRGB(255,255,255) or Color3.fromRGB(180,180,180)
    end
end

-- منطق البحث
local function searchPlayer(query)
    query = (query or ""):gsub("^%s+",""):gsub("%s+$",""):lower()
    if #query == 0 then
        foundTarget=nil; ResultLbl.Text="لم يُحدد لاعب"
        ResultLbl.TextColor3=Color3.fromRGB(140,140,140); AvatarCircle.Image=""
        return
    end
    local best, bestScore = nil, -1
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp then
            local n=p.Name:lower(); local dn=p.DisplayName:lower(); local id=tostring(p.UserId)
            local score=0
            if n==query or dn==query or id==query then score=3
            elseif n:sub(1,#query)==query or dn:sub(1,#query)==query then score=2
            elseif n:find(query,1,true) or dn:find(query,1,true) or id:find(query,1,true) then score=1
            end
            if score>bestScore then bestScore=score; best=p end
        end
    end
    if best then
        foundTarget=best
        ResultLbl.Text="  "..best.DisplayName.." @"..best.Name
        ResultLbl.TextColor3=Color3.fromRGB(255,255,255)
        local ok,img=pcall(Players.GetUserThumbnailAsync,Players,
            best.UserId,Enum.ThumbnailType.HeadShot,Enum.ThumbnailSize.Size100x100)
        if ok then AvatarCircle.Image=img end
        showNotif("تم تحديد: "..best.DisplayName)
    else
        foundTarget=nil; ResultLbl.Text="لم يُعثر على لاعب"
        ResultLbl.TextColor3=Color3.fromRGB(200,70,70); AvatarCircle.Image=""
    end
end
SearchBox:GetPropertyChangedSignal("Text"):Connect(function() searchPlayer(SearchBox.Text) end)

local function setSpeed(v)
    bodyMoveSpeed   = math.clamp(math.floor(v), MIN_SPD, MAX_SPD)
    SpeedLbl.Text   = "سرعة الحركة: "..bodyMoveSpeed
    local ratio     = (bodyMoveSpeed-MIN_SPD)/(MAX_SPD-MIN_SPD)
    SliderFill.Size = UDim2.new(ratio,0,1,0)
    SliderKnob.Position = UDim2.new(ratio,-7,0.5,-7)
    SpeedInput.Text = tostring(bodyMoveSpeed)
    -- تطبيق السرعة على WalkSpeed مباشرة
    local c = lp.Character
    if c then
        local h = c:FindFirstChildOfClass("Humanoid")
        if h then
            h.WalkSpeed = bodyMoveSpeed
        end
    end
end

do
    local holding=false
    SliderKnob.InputBegan:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseButton1 then holding=true end
    end)
    UIS.InputEnded:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseButton1 then holding=false end
    end)
    UIS.InputChanged:Connect(function(inp)
        if holding and inp.UserInputType==Enum.UserInputType.MouseMovement then
            local ap=SliderTrack.AbsolutePosition; local as=SliderTrack.AbsoluteSize
            local ratio=math.clamp((inp.Position.X-ap.X)/as.X,0,1)
            setSpeed(MIN_SPD+ratio*(MAX_SPD-MIN_SPD))
        end
    end)
end
SpeedInput.FocusLost:Connect(function() local v=tonumber(SpeedInput.Text); if v then setSpeed(v) end end)

end

local StartFlyBtn = makeFlyBtn("🛫 طيران", 8,  Color3.fromRGB(0,60,160))
local StopFlyBtn  = makeFlyBtn("🛬 إيقاف", 118, Color3.fromRGB(0,30,80))

-- منطق الطيران — كاميرا حرة (شيفت لوك)
-- الشخصية تراعي وين ما الكاميرا تراعي، بدون تحديد أزرار
local UserGameSettings = UserSettings():GetService("UserGameSettings")
local prevRotationType = Enum.RotationType.MovementRelative  -- افتراضي

local function stopFly()
    isFlying = false
    if flyRSConn then flyRSConn:Disconnect(); flyRSConn = nil end
    -- تدمير كل physics objects قديمة
    if flyBodyVelocity and flyBodyVelocity.Parent then
        pcall(function() flyBodyVelocity.Parent:FindFirstChild("FlyAttach0") and flyBodyVelocity.Parent.FlyAttach0:Destroy() end)
        flyBodyVelocity:Destroy()
    end
    if flyBodyGyro and flyBodyGyro.Parent then flyBodyGyro:Destroy() end
    flyBodyVelocity = nil; flyBodyGyro = nil
    pcall(function() UserGameSettings.RotationType = prevRotationType end)
    local c = lp.Character
    if c then
        local h = c:FindFirstChildOfClass("Humanoid")
        if h then h.PlatformStand = false end
        -- تنظيف أي attachments متبقية
        local root = c:FindFirstChild("HumanoidRootPart")
        if root then
            for _, v in pairs(root:GetChildren()) do
                if v.Name == "FlyAttach0" or v:IsA("BodyVelocity") or
                   v:IsA("BodyGyro") or v:IsA("LinearVelocity") or
                   v:IsA("AlignOrientation") then
                    pcall(function() v:Destroy() end)
                end
            end
        end
    end
    StartFlyBtn.BackgroundColor3 = Color3.fromRGB(0,60,160)
end

local function startFly()
    if isFlying then return end
    local c = lp.Character; if not c then return end
    local h = c:FindFirstChildOfClass("Humanoid"); if not h then return end
    local root = c:FindFirstChild("HumanoidRootPart"); if not root then return end

    -- تنظيف أي physics قديمة
    for _, v in pairs(root:GetChildren()) do
        if v:IsA("BodyVelocity") or v:IsA("BodyGyro") or
           v:IsA("LinearVelocity") or v:IsA("AlignOrientation") or
           v.Name == "FlyAttach0" then
            pcall(function() v:Destroy() end)
        end
    end

    pcall(function()
        prevRotationType = UserGameSettings.RotationType
        UserGameSettings.RotationType = Enum.RotationType.CameraRelative
    end)

    isFlying = true
    h.PlatformStand = true
    StartFlyBtn.BackgroundColor3 = Color3.fromRGB(0,120,0)

    -- جرّب LinearVelocity الجديدة أولاً، وإلا BodyVelocity قديمة
    local useNewAPI = false
    local att0 = Instance.new("Attachment")
    att0.Name = "FlyAttach0"
    att0.Parent = root

    local ok = pcall(function()
        local lv = Instance.new("LinearVelocity")
        lv.MaxForce       = math.huge
        lv.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
        lv.RelativeTo     = Enum.ActuatorRelativeTo.World
        lv.VectorVelocity = Vector3.zero
        lv.Attachment0    = att0
        lv.Parent         = root
        flyBodyVelocity   = lv
        useNewAPI         = true
    end)

    if not ok then
        att0:Destroy()
        local bv = Instance.new("BodyVelocity")
        bv.Velocity  = Vector3.zero
        bv.MaxForce  = Vector3.new(4e5, 4e5, 4e5)
        bv.P         = 2e4
        bv.Parent    = root
        flyBodyVelocity = bv
    end

    flyRSConn = RunService.Heartbeat:Connect(function()
        if not isFlying then return end
        local rc = lp.Character; if not rc then stopFly(); return end
        local rr = rc:FindFirstChild("HumanoidRootPart"); if not rr then stopFly(); return end
        local rh = rc:FindFirstChildOfClass("Humanoid"); if not rh then return end

        -- ابقَ في وضع PlatformStand دائماً (اللعبة قد تعيده)
        rh.PlatformStand = true

        local cam     = workspace.CurrentCamera
        local lookDir = cam.CFrame.LookVector

        if flyBodyVelocity and flyBodyVelocity.Parent then
            if useNewAPI then
                flyBodyVelocity.VectorVelocity = lookDir * flySpeed
            else
                flyBodyVelocity.Velocity = lookDir * flySpeed
            end
        end

        -- واجّه الشخصية لاتجاه الطيران (أفقي فقط حتى لا تميل)
        local flatLook = Vector3.new(lookDir.X, 0, lookDir.Z)
        if flatLook.Magnitude > 0.01 then
            rr.CFrame = CFrame.new(rr.Position, rr.Position + flatLook)
                        * CFrame.Angles(math.asin(math.clamp(lookDir.Y,-1,1)), 0, 0)
        end
    end)
end

StartFlyBtn.MouseButton1Click:Connect(function()
    if isFlying then stopFly() else startFly() end
end)
StopFlyBtn.MouseButton1Click:Connect(function() stopFly() end)
